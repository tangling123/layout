# 08 接口草案（声明级契约）

> 只列**接口与关键类型**，不含实现。目的是让各模块能并行开发：接口冻结后，
> `core/io/scene/render/ui` 可以用 `FakeReader`、`FakeScene` 分别推进。
> 命名遵循 06 的编码规范；`Q_OBJECT` 类只出现在 `ui/app` 层。

## 1. core

```cpp
namespace eda::core {

// ---- 错误 ----
enum class ErrorCode { Ok, FileNotFound, PermissionDenied, UnknownFormat,
                       UnsupportedVersion, CorruptRecord, TruncatedFile,
                       TooLarge, OutOfMemory, Cancelled,
                       WriteFailed, ReadOnlyDocument, InvalidEdit, CycleDetected,
                       Internal };

struct Error {
    ErrorCode      code = ErrorCode::Ok;
    int64_t        offset = -1;     // 文件偏移（若相关）
    std::string    message;
    std::string    context;         // cell 名 / 记录名
};

// C++20 兼容：用轻量 Result（C++23 可直接换 std::expected）
template <class T>
class Result {
public:
    static Result ok(T value);
    static Result fail(Error e);
    explicit operator bool() const;      // true = 成功
    T&        value();                   // 失败时断言
    const T&  value() const;
    const Error& error() const;
};

struct Warning {
    int         code = 0;
    int64_t     offset = -1;
    std::string cell;
    std::string message;
};

using WarningSink = std::function<void(const Warning&)>;

// ---- 单位 ----
class Units {
public:
    double dbuPerUserUnit = 1000.0;
    double metersPerDbu    = 1e-9;
    bool   valid          = false;

    constexpr double micronsPerDbu() const noexcept { return metersPerDbu * 1e6; }
    double toMicrons(int64_t dbu) const noexcept;
    int64_t fromMicrons(double um) const noexcept;
    bool operator==(const Units&) const noexcept;
};

// ---- 几何 ----
struct Point { int32_t x = 0, y = 0; };
struct Box {
    Point min, max;
    bool isEmpty() const noexcept;
    bool contains(Point p) const noexcept;
    bool intersects(const Box& b) const noexcept;
    Box  united(const Box& b) const noexcept;
    int64_t width() const noexcept;
    int64_t height() const noexcept;
};

class Transform {
public:
    // p' = M*p + t，M = [a c; b d]（列向量约定，与 04 §3.1 一致）
    double a = 1, b = 0, c = 0, d = 1;
    double tx = 0, ty = 0;              // DBU
    bool   mirrored = false;
    double mag = 1.0;
    double angleDeg = 0.0;

    bool isIdentity() const noexcept;
    Point  apply(Point p) const noexcept;      // 内部 long double，最后 llround
    Box    apply(const Box& b) const noexcept; // 四角变换取包围盒
    Transform inverse() const noexcept;        // 用于拾取反变换
    static Transform compose(const Transform& parent, const Transform& local);
    static Transform fromGdsStrans(bool mirror, double mag, double angleDeg,
                                   Point offset);
};

// ---- 层 ----
struct LayerKey {
    uint32_t layer    = 0;
    uint32_t datatype = 0;
    uint32_t nameId   = 0;   // 0 = 无名（GDS 恒为 0）
    auto operator<=>(const LayerKey&) const = default;
};

// ---- 模型（可变；写入只经 Transaction） ----
enum class ShapeKind { Boundary, Box, Path, Text, Node, XElement };

using CellId     = uint32_t;          // 稳定，不复用
using ElementKey = uint64_t;          // index(低 32) | generation(高 32)

constexpr CellId kInvalidCell = 0;

// 带 generation 的槽位容器：删除后旧句柄自动失效
template <class T>
class SlotMap {
public:
    ElementKey insert(T value);
    T*          find(ElementKey k);            // generation 不匹配 → nullptr
    const T*    find(ElementKey k) const;
    bool        erase(ElementKey k);
    template <class F> void forEach(F&& f) const;
    size_t      size() const;
    uint64_t    revision() const;              // 结构变更计数
};

struct Shape {
    ShapeKind         kind = ShapeKind::Boundary;
    LayerKey          layer;
    Box               bbox;                    // cell 局部
    std::vector<Point> points;                 // Boundary / Path / Box(2 点)
    int32_t           width = 0;               // Path
    uint8_t           endType = 0;             // Path: flush/round/extend/custom
    Point             anchor;                  // Text
    uint32_t          stringId = 0;            // Text
    int32_t           heightDbu = 0;           // Text
    uint16_t          presentation = 0;        // Text
    std::vector<uint8_t> raw;                  // Node / XElement（原样回写）
};

struct Instance {
    CellId      target = kInvalidCell;
    Transform   xf;
    Repetition  rep;                           // 单实例时 rep.count()==1
};

struct ElementRef {                            // 跨模块唯一的元素引用
    CellId     cell = kInvalidCell;
    ElementKey key{};
    bool       isInstance = false;
    auto operator<=>(const ElementRef&) const = default;
};

// 吸附的候选来源（由 scene 实现、editing 使用；放在 core 避免 scene↔editing 互相依赖）
class ISnapSource {
public:
    struct Candidate { Point point; bool isVertex = false; bool isInstanceOrigin = false; };
    virtual ~ISnapSource() = default;
    virtual void nearby(Point world, int64_t radiusDbu,
                        std::vector<Candidate>& out) const = 0;
};

struct Cell {
    CellId                id = kInvalidCell;
    uint32_t              nameId = 0;
    Box                   bboxLocal;
    uint64_t              revision = 0;        // 内容版本
    std::vector<CellId>   parents;             // 反向引用
    SlotMap<Shape>        shapes;
    SlotMap<Instance>     instances;
    std::vector<Property> props;
};

// ---- 事务（撤销的最小单位） ----
enum class ChangeKind { InsertShape, ReplaceShape, EraseShape,
                        InsertInstance, ReplaceInstance, EraseInstance,
                        CreateCell, RenameCell, DeleteCell,
                        AddLayer, SetLayerName };

struct Change {
    ChangeKind  kind;
    CellId      cell = kInvalidCell;
    ElementKey  key{};
    Shape       shapeBefore, shapeAfter;
    Instance    instBefore, instAfter;
    std::string strBefore, strAfter;
};

class Transaction {                            // 构造仅由 Document::beginTransaction 发起
public:
    // 非公开构造（此处为草案展示）：外部只能拿到 unique_ptr<Transaction>
    Transaction(Layout& layout, std::string label);
    ~Transaction();                            // 未 commit → 自动 rollback

    Result<ElementKey> insertShape(CellId, Shape);
    Result<void>       replaceShape(CellId, ElementKey, Shape);
    Result<void>       eraseShape(CellId, ElementKey);
    Result<ElementKey> insertInstance(CellId, Instance);
    Result<void>       setInstance(CellId, ElementKey, Instance);
    Result<void>       eraseInstance(CellId, ElementKey);
    Result<CellId>     createCell(std::string name);
    Result<void>       renameCell(CellId, std::string name);
    Result<void>       deleteCell(CellId);

    Result<void> commit();                     // 追加到父事务或 EditHistory
    void         rollback();
    bool         isEmpty() const;
    const std::string& label() const;
    std::span<const CellId> dirtyCells() const;
};

class EditHistory {
public:
    void push(std::unique_ptr<Transaction> txn);
    bool canUndo() const;
    bool canRedo() const;
    Result<void> undo(Layout&);
    Result<void> redo(Layout&);
    std::string undoLabel() const;
    std::string redoLabel() const;
    void markSaved();
    bool isAtSavePoint() const;
    void setMemoryLimit(size_t bytes);         // 默认 512 MB
    void setStepLimit(size_t steps);           // 默认 200
    size_t memoryUsage() const;
};

class Layout {
public:
    // ---- 只读 ----
    const std::string& name() const;
    const Units& units() const;
    const std::filesystem::path& path() const;
    std::span<const Cell> cells() const;
    const Cell* cell(CellId id) const;
    const Cell* cellByName(std::string_view name) const;
    CellId findCell(std::string_view name) const;
    std::string_view string(uint32_t id) const;    // StringPool
    std::span<const LayerKey> layers() const;
    const Box& bbox() const;                       // 全库
    const Stats& stats() const;
    uint64_t globalRevision() const;

    // ---- 写入：不公开 ----
    // Transaction 以 friend 身份调用 Layout 的私有写方法（insertShape/replaceShape/…），
    // 因此外部永远拿不到「可写 Layout」；这条约束是撤销与脏追踪正确性的前提。
    // 写方法内部维护：环路检查、parents 同步、bbox 上溯、dirty 记录、revision++。

private:
    friend class Transaction;
};

} // namespace eda::core

namespace eda::io {

class LayerMap;   // 定义见下（.lyp / CSV 层映射）

struct IOOptions {
    bool     strict = false;             // true: 遇到可疑数据立即失败
    uint64_t maxElements = 0;            // 0 = 不限

    bool     loadNode    = false;
    bool     loadProperties = true;
    std::optional<core::Units> unitsOverride;
};

struct Progress {
    std::atomic<bool> cancelled{false};
    int64_t totalBytes = 0;
    int64_t doneBytes  = 0;
    std::string phase;                   // "scan" / "parse" / "index"
    std::function<void(int64_t done, int64_t total)> onUpdate;  // 每 64KB 调用
};

struct ReadStats {
    uint64_t cells = 0, elements = 0, polygons = 0, paths = 0,
             texts = 0, srefs = 0, arefs = 0;
    int64_t  parseMs = 0;
    uint32_t warnings = 0;
};

struct WriteOptions {
    std::string format;                  // "gds" / "oasis"；空 = 按扩展名
    bool     compress       = true;      // OASIS CBLOCK
    bool     writeOffsetTable = true;    // OASIS END 后的 offset table
    bool     keepRawRecords = true;      // NODE/未知记录原样回写
    bool     splitLargePolygons = true;  // GDS 8191 点切分
    bool     verifyAfterWrite = true;    // 写后自检（计数一致）
    std::shared_ptr<const LayerMap> layerMap;  // 可选：改写层号/名字
    std::optional<core::Units> unitsOverride;
};

struct WriteStats {
    uint64_t cells = 0, elements = 0;
    int64_t  bytes = 0, writeMs = 0;
    uint32_t warnings = 0;
};

// ---------------------------------------------------------------------------
// 图层属性与映射：.lyp 是权威格式（KLayout 兼容），CSV/layermap 仅作导入
// ---------------------------------------------------------------------------
struct LayerProperties {
    core::LayerKey key;

    bool           visible    = true;
    int            zOrder     = 0;
    std::string    group;             // 层组名（KLayout group）

// 读写策略：内部保留一份「原始 XML DOM」，写回时只改对应节点的 value，
// 未知元素/属性原样保留（工艺厂的 .lyp 常有自定义字段，不能洗掉）。
class LayerMap {
public:
    static core::Result<LayerMap> loadLyp(const std::filesystem::path&);
    static core::Result<LayerMap> loadCsv(const std::filesystem::path&);

    std::span<const LayerProperties> properties() const;
    std::optional<LayerProperties>   find(core::LayerKey) const;
    void set(const LayerProperties&);                 // 新增或覆盖
    void setVisible(core::LayerKey, bool);
    void setColor(core::LayerKey, uint32_t fill, uint32_t frame);

    bool isFromLyp() const;
    bool dirty() const;
    core::Result<void> saveLyp(const std::filesystem::path&) const;  // 原子写由调用方负责
    core::Result<void> saveCsv(const std::filesystem::path&) const;
};

// 所有格式的统一入口
class FormatReader {
public:
    virtual ~FormatReader() = default;
    virtual std::string_view name() const = 0;                    // "gds" / "oasis"
    virtual std::span<const std::string> extensions() const = 0;  // {".gds"} ...

    virtual core::Result<void> open(const std::filesystem::path& p) = 0;

    // 主流程：一次性解析出完整 Layout（MVP）；后续可加 lazy 版本
    virtual core::Result<ReadStats> read(core::Layout& out,
                                         const IOOptions& opt,
                                         Progress& progress,
                                         core::WarningSink sink) = 0;

    // 可选能力（不实现时返回 false / Unsupported）
    virtual bool supportsLazy() const { return false; }
    virtual core::Result<void> readCell(core::Layout&, core::CellId,

        return core::Result<void>::fail({core::ErrorCode::Internal, -1,
                                         "lazy loading not supported", {}});
    }
};

public:
    using ReaderFactory = std::function<std::unique_ptr<FormatReader>()>;
    using WriterFactory = std::function<std::unique_ptr<FormatWriter>()>;
    void registerFormat(std::string_view name, ReaderFactory r, WriterFactory w);
    std::unique_ptr<FormatReader> createReader(std::string_view name);
    std::unique_ptr<FormatWriter> createWriter(std::string_view name);
    // 后缀 + 魔数（GDS 无魔数 → 检查首记录；OASIS → "%SEMI"+"OASIS"）
    core::Result<std::unique_ptr<FormatReader>>
        sniff(const std::filesystem::path& p);
    // 保存时的格式推断（扩展名 > 当前文档格式 > 报错）
    core::Result<std::string> formatForSave(const std::filesystem::path& p,
                                            std::string_view currentFormat);
};

// 写出：只读 Layout，写到目标路径（调用方保证编辑已锁定）
class FormatWriter {
public:
    virtual ~FormatWriter() = default;
    virtual std::string_view name() const = 0;
    virtual std::span<const std::string> extensions() const = 0;

    virtual core::Result<WriteStats> write(const core::Layout& layout,
                                          const std::filesystem::path& target,
                                          const WriteOptions& opt,
                                          Progress& progress,
                                          core::WarningSink sink) = 0;
};

class StreamReader {   // 大端读取 + 偏移 + 越界检测
public:
    explicit StreamReader(std::span<const std::byte> data);
    uint8_t  u8();
    uint16_t u16();
    uint32_t u32();
    int16_t  i16();
    int32_t  i32();
    double   f64();                       // IEEE754
    std::string_view str(size_t n);
    bool     eof() const;
    int64_t  offset() const;              // 当前读位置
    core::Result<void> seek(int64_t off);
    uint64_t uvarint();                   // OASIS
    int64_t  svarint();                   // OASIS（符号在最低位）
};

} // namespace eda::io
```

**约定**

- `read()` 只往传入的 `Layout` 写入，不保留文件句柄；失败时内容为「已成功解析的部分」。
- `read()` 内部若需要构造复杂结构，先写临时 `Layout`，成功后由调用方 `move` 进 `Document`。
- `write()` 接收 `const Layout&`；调用方负责 `Document::lockEditing()`（见 09 §7.1）。
- 写出必须自检：`WriteOptions::verifyAfterWrite`（默认开）比较 cell/元素计数，不一致返回
  `WriteFailed` 并保留临时文件供排查。
- `Progress` 由调用方（`TaskRunner`）创建，解析器/写入器只更新计数并检查 `cancelled`。
- `WarningSink` 允许为空指针；场景层与 UI 都复用它。

namespace eda::scene {

using core::CellId;
using core::ElementKey;
using core::Instance;
using core::ElementRef;

struct LodOptions {
    double cullPx        = 1.5;
    double boxPx         = 8.0;
    double simplifyPx    = 32.0;
    uint32_t elementBudget = 1'500'000;   // 单帧图元预算
    uint32_t instanceBudget = 200'000;
    uint32_t maxDepth    = 64;
    uint32_t maxNodes    = 5'000'000;

struct SceneStats {
    uint64_t cells = 0, shapes = 0, instances = 0, vertices = 0;
    uint32_t maxDepth = 0;
    uint64_t sceneBytes = 0;      // 估算
    uint32_t warnings = 0;
};

enum class DrawMode { Cull, Box, Geometry, Recurse };

// scene → render 的顶点契约（POD，16 B；render 层直接上传 GPU）
struct LayoutVertex {
    float    x, y;        // cell 局部坐标（µm）
    uint32_t rgba;        // 层颜色 + alpha
    uint32_t elementSlot; // 拾取用：slot 低 32 位（+ isInstance 位）
};

struct LayerBatch {
    core::LayerKey        layer;
    std::vector<LayoutVertex> vertices;
    std::vector<uint32_t> indices;
    core::Box             localBox;
    std::vector<ElementKey> elementKeys;   // 顶点/索引 → 元素句柄（偏移表）
};
// 注：顶点为 cell 局部 µm（SceneBuilder 用 Units 从 DBU 换算一次）；
//     Instance::xf 保留 DBU 语义供属性显示/精确计算，
//     另在构建期派生 µm 空间的 modelMatrix（float[12]）供绘制使用。

class CellGeometry {          // 只读，可被所有实例共享；GPU 缓存以它为键
public:
    CellId cell() const;
    std::span<const LayerBatch> layers() const;
    std::span<const InstanceRef> instances() const;   // 递归用
    core::Box localBox() const;
    uint64_t elementCount() const;
    uint64_t revision() const;                        // 构建时的 cell.revision
    // 拾取：局部坐标 + 容差 → 候选（返回元素句柄）
    std::optional<ElementKey> hitTestLocal(core::Point p, double tolDbu,
                                          const LayerVisibility& vis) const;
};

struct InstanceRef {          // CellGeometry 内的实例条目（不含路径）
    core::Instance inst;
    core::Box      childBoxLocal;    // 变换后的目标 bbox（cell 局部）
};

// 层可见性抽象：由 render::LayerStyleTable 实现，避免 scene 依赖 render

public:

    virtual bool isVisible(const core::LayerKey& key) const = 0;

public:

    const core::Box& worldBoxMicrons() const;             // 便于渲染/相机
    std::span<const CellId> cellIds() const;
    const CellGeometry* geometry(CellId c) const;         // 未构建 → nullptr

    const SceneStats& stats() const;
    uint64_t generation() const;                          // 发布序号

    // 渲染层唯一入口：从 topCell 递归，visitor(cellId, geometry, modelMatrix, depth)
    // LOD 决策在 render 层用 LodOptions 完成（scene 不依赖相机类型）
    template <class Visitor>
    void visit(const core::Box& worldBox, const LodOptions&, Visitor&& visitor) const;

    // 增量发布：只替换脏条目，其余 shared_ptr 复用

    };
    Result apply(const core::Layout& layout,

## 7. 接口冻结检查表（写代码前过一遍）

- [ ] `Units` / `Transform` 的约定与 04 文档完全一致（列向量、mirror 顺序）。
- [ ] `core::Layout` 的**写接口只对 `Transaction` 开放**（friend / 访问控制），外部无法绕过事务。
- [ ] `SceneGraph` 是**不可变**的，渲染层只读；编辑后产生新快照（复用未变 `CellGeometry`）。
- [ ] `IRenderer::sync` 不做几何生成，只做「CPU→GPU」传输与缓存管理。
- [ ] `DocumentController`（视图）与 `EditController`（数据）职责分离，UI 面板不直接改 `Layout`。
- [ ] UI 只通过 `CommandRouter::invoke(命令ID)` 改状态；`EditController::apply` 对 UI 不可见（11 §0）。
- [ ] 每个命令 ID 唯一、带版本、在 `command_table.inc` 有且仅有一行；每条 undoable 命令有 undo 用例。
- [ ] `command` 层不出现 Qt 类型与 `scene/render/ui/app` 头文件；文本编解码只在 `CommandCodec`。
- [ ] `ElementRef` = `{cell, ElementKey, isInstance}`，不依赖指针；句柄删除后自动失效。
- [ ] `Transaction` 失败路径必须回滚；`commit()` 之后才进 `EditHistory`。
- [ ] `TaskRunner` 的取消语义：任务返回 `Cancelled`，且不再回投成功结果。
- [ ] 所有公开头文件在 `core` / `editing` 层不出现 `QString/QColor`（Qt 只出现在 render/ui/app）。
- [ ] `save()` 的默认路径是**原地覆盖**：临时文件同目录 + `fsync` + `rename`，失败时原文件必须完好。
- [ ] 保存前检查 `Document::wasExternallyModified()`；冲突必须由用户选择，禁止静默覆盖。
- [ ] `io::LayerMap` 写回 `.lyp` 时保留未知 XML 元素；图层属性改动不进 `EditHistory`。
- [ ] `scene` 只依赖 core/io；`editing` 只依赖 core/io；两者之间只通过 `core` 中的接口/值类型通信。
